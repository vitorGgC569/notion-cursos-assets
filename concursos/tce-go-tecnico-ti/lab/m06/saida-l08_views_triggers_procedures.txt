Triggers: auditoria automatica (PostgreSQL com PL/pgSQL)
======================================================================
pg> CREATE FUNCTION audita() RETURNS trigger LANGUAGE plpgsql AS $$
   BEGIN
     IF TG_OP = 'UPDATE' THEN
       INSERT INTO auditoria (quando, operacao, numero, antes, depois) VALUES ('t', TG_OP, OLD.numeroprocesso, OLD.tribunalprocesso, NEW.tribunalprocesso);
       RETURN NEW;
     ELSIF TG_OP = 'DELETE' THEN
       INSERT INTO auditoria (quando, operacao, numero, antes, depois) VALUES ('t', TG_OP, OLD.numeroprocesso, OLD.tribunalprocesso, NULL);
       RETURN OLD;
     END IF;
     RETURN NEW;
   END $$
OK
pg> CREATE TRIGGER trg_audita AFTER UPDATE OR DELETE ON processo FOR EACH ROW EXECUTE FUNCTION audita()
OK
pg> UPDATE processo SET tribunalprocesso = 4 WHERE numeroprocesso = '000467'
OK (1 linha afetada)
pg> DELETE FROM processo WHERE numeroprocesso = '000999'
OK (1 linha afetada)
pg> SELECT operacao, numero, antes, depois FROM auditoria ORDER BY id
operacao | numero | antes | depois
---------+--------+-------+-------
UPDATE   | 000467 | 5     | 4
DELETE   | 000999 | 2     | NULL
(2 linhas)

Trigger BEFORE: valida e pode alterar a linha antes de gravar
pg> CREATE FUNCTION valida_ano() RETURNS trigger LANGUAGE plpgsql AS $$
   BEGIN
     IF NEW.anoinicioprocesso < 1980 THEN RAISE EXCEPTION 'ano invalido: %', NEW.anoinicioprocesso; END IF;
     NEW.numeroprocesso := upper(NEW.numeroprocesso);
     RETURN NEW;
   END $$
OK
pg> CREATE TRIGGER trg_valida BEFORE INSERT ON processo FOR EACH ROW EXECUTE FUNCTION valida_ano()
OK

(1 linha)
pg> CREATE PROCEDURE arquiva_lote() LANGUAGE plpgsql AS $$
   BEGIN
     INSERT INTO log_lote SELECT COUNT(*) FROM processo;
     COMMIT;
     INSERT INTO log_lote SELECT COUNT(*) FROM processo WHERE anoinicioprocesso > 2010;
     COMMIT;
   END $$
OK

OK

-

(2 linhas)
